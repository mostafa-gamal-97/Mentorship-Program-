\timing

-- ================================
-- Query 5: Total number of products per category
-- ================================
-- Before optimization
RAISE NOTICE 'Query 5 Before Optimization';
EXPLAIN ANALYZE
SELECT
    c.category_id,
    c.category_name,
    COUNT(p.product_id) AS products_count_per_category
FROM categories c
LEFT JOIN products p
    ON p.category_id = c.category_id
GROUP BY c.category_id, c.category_name
ORDER BY c.category_id;

-- ================================
-- Query 6: Top customers by total spending
-- ================================
RAISE NOTICE 'Query 6 Before Optimization';
EXPLAIN ANALYZE
SELECT
    c.customer_id,
    c.first_name,
    c.last_name,
    SUM(o.total_amount) AS total_spent
FROM customers c
JOIN orders o
    ON o.customer_id = c.customer_id
GROUP BY c.customer_id, c.first_name, c.last_name
ORDER BY total_spent DESC;

-- ================================
-- Query 7: Most recent 1000 orders
-- ================================
RAISE NOTICE 'Query 7 Before Optimization';
EXPLAIN ANALYZE
SELECT
    c.customer_id,
    c.first_name,
    c.last_name,
    o.order_id,
    o.order_date
FROM customers c
JOIN orders o
    ON c.customer_id = o.customer_id
ORDER BY o.order_date DESC
LIMIT 1000;

-- ================================
-- Query 8: Products with stock < 10
-- ================================
RAISE NOTICE 'Query 8 Before Optimization';
EXPLAIN ANALYZE
SELECT
    product_id,
    name,
    stock_quantity
FROM products
WHERE stock_quantity < 10
ORDER BY product_id;

-- ================================
-- Query 9: Revenue per category
-- ================================
RAISE NOTICE 'Query 9 Before Optimization';
EXPLAIN ANALYZE
SELECT
    c.category_id,
    c.category_name,
    SUM(od.quantity * od.unit_price) AS total_revenue
FROM categories c
JOIN products p
    ON p.category_id = c.category_id
JOIN order_details od
    ON od.product_id = p.product_id
GROUP BY c.category_id, c.category_name
ORDER BY total_revenue DESC;

-- ================================
-- Create Indexes for Optimization
-- ================================
RAISE NOTICE 'Creating Indexes for Optimization';

-- Query 5 & 9
CREATE INDEX IF NOT EXISTS idx_products_category_id ON products(category_id);

-- Query 6
CREATE INDEX IF NOT EXISTS idx_orders_customer_id ON orders(customer_id);

-- Query 7
CREATE INDEX IF NOT EXISTS idx_orders_order_date_desc ON orders(order_date DESC);

-- Query 8 (Partial Index)
CREATE INDEX IF NOT EXISTS idx_products_low_stock ON products(product_id, stock_quantity)
WHERE stock_quantity < 10;

-- Query 9
CREATE INDEX IF NOT EXISTS idx_order_details_product_id ON order_details(product_id);
