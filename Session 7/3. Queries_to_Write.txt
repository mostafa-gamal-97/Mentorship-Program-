-- 5. Write a SQL Query to Retrieve the total number of products in each category.
EXPLAIN ANALYZE SELECT
    c.category_id,
    c.category_name,
    COUNT(p.product_id) AS products_count_per_category
FROM categories c
LEFT JOIN products p
    ON p.category_id = c.category_id
GROUP BY
    c.category_id,
    c.category_name
ORDER BY
    c.category_id;

-- 6. Write a SQL Query to Find the top customers by total spending.
EXPLAIN ANALYZE SELECT
    c.customer_id,
    c.first_name,
    c.last_name,
    SUM(o.total_amount) AS total_spent
FROM customers c
JOIN orders o
ON o.customer_id = c.customer_id
GROUP BY
    c.customer_id,
    c.first_name,
    c.last_name
ORDER BY
    total_spent DESC;

-- 7. Write a SQL Query to Retrieve the most recent 1000 orders with customer information.
EXPLAIN ANALYZE SELECT
    c.customer_id,
    c.first_name,
    c.last_name,
    o.order_id,
    o.order_date
FROM customers c
JOIN orders o
ON c.customer_id = o.customer_id
ORDER BY o.order_date DESC
LIMIT 1000;

-- 8. Write a SQL Query to List products with stock quantity of less than 10.
EXPLAIN ANALYZE SELECT
    product_id,
    name,
    stock_quantity
FROM products
WHERE stock_quantity < 10
ORDER BY product_id;

-- 9. Write a SQL Query to Calculate the revenue generated from each category.
EXPLAIN ANALYZE SELECT
    c.category_id,
    c.category_name,
    SUM(od.quantity * od.unit_price) AS total_revenue
FROM categories c
JOIN products p
ON p.category_id = c.category_id
JOIN order_details od
ON od.product_id = p.product_id
GROUP BY
    c.category_id,
    c.category_name
ORDER BY
    total_revenue DESC;
